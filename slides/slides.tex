\subsection{some numerics}

\begin{frame}{discretize the SKE}
\begin{center}
\includegraphics[width=0.5\textwidth]{surfacenotation}
\end{center}

\begin{itemize}
\item planar (flow-line) SKE: \qquad $\displaystyle \frac{\partial s}{\partial t} + u \frac{\partial s}{\partial x} - w = a$

\smallskip
\item choose time step $\Delta t>0$ and grid spacing $\Delta x>0$
\item approximate partial derivatives by finite difference\footnote{helpful textbooks include LeVeque \cite{Leveque2007} and Morton \& Mayers \cite{MortonMayers2005}} quotients:
    $$\frac{\partial s}{\partial t} \approx \frac{\snew_j - s_j}{\Delta t}, \qquad \frac{\partial s}{\partial x} \approx \frac{s_{j+1} - s_{j-1}}{2\Delta x}$$
\phantom{foo}
\end{itemize}
\end{frame}


\begin{frame}{discretize the SKE}
\begin{center}
\includegraphics[width=0.5\textwidth]{surfacenotation}
\end{center}

\begin{itemize}
\item the SKE becomes discrete:
    $$\frac{\snew_j - s_j}{\Delta t} + u_j \frac{s_{j+1} - s_{j-1}}{2\Delta x} - w_j = a_j$$
\item write as an \alert{update} equation which determines $\snew_j$:
	$$\snew_j = s_j - \Delta t\, u_j \frac{s_{j+1}-s_{j-1}}{2\Delta x} + \Delta t (a_j + w_j)$$
\end{itemize}
\end{frame}


\begin{frame}{upwinding}

\begin{itemize}
\item however, if we implement this explicit ``forward-time centered-space'' scheme
	$$\snew_j = s_j - \Delta t\, u_j \frac{s_{j+1}-s_{j-1}}{2\Delta x} + \Delta t (a_j + w_j) \hspace{1.0cm} \leftarrow \text{\emph{bad}}$$
then unstable things happen! (exercise \textbf{2})
\item instead, it is known that an explicit \alert{upwind} scheme is stable
    \begin{itemize}
    \item[$\circ$] \alert{conditionally} stable, anyway, for this kind of \alert{advection} problem
    \end{itemize}

\item in the case when the velocity is positive ($u_j \ge 0$) upwind is:
	$$\hspace{3mm} \snew_j = s_j - \Delta t\, u_j \frac{s_j-s_{j-1}}{\Delta x} + \Delta t (a_j + w_j) \hspace{1.2cm} \leftarrow \text{\emph{useful}}$$

    \begin{itemize}
    \item[$\circ$] general upwind formula: \quad $\displaystyle \frac{\partial s}{\partial x} \approx \frac{1}{\Delta x} \begin{cases} (s_j-s_{j-1}), & u_j \ge 0 \\ (s_{j+1}-s_j), & u_j < 0 \end{cases}$
    \item[$\circ$] the condition for stability is known (more soon)
    \end{itemize}
%\item see textbooks for these caveats
\end{itemize}
\end{frame}


\begin{frame}[fragile]\frametitle{implementation}

\begin{itemize}
\item assuming $u_j \ge 0$, the upwind scheme for the SKE is
    $$\snew_j = s_j - \Delta t\, u_j \frac{s_j-s_{j-1}}{\Delta x} + \Delta t (a_j + w_j)$$
\item this becomes a Python function:
\begin{lstlisting}[language=PythonPlus]
def explicitupwindstep(x, s, dt):
    dx = x[1] - x[0]
    snew = s.copy()
    snew[1:] -= dt * u(x[1:]) * (s[1:] - s[:-1]) / dx
    snew[1:] += dt * (a(x[1:]) + w(x[1:]))
    return snew
\end{lstlisting}
    \begin{itemize}
    \item[$\circ$] \texttt{x}, \texttt{s}, \texttt{snew} are 1D NumPy arrays
    \item[$\circ$] \texttt{x} is an equally-spaced grid
    \item[$\circ$] note that separate functions define $a(x)$, $u(x)$, $w(x)$
    \item[$\circ$] note that \, \texttt{snew[0] = s[0]} \, is never changed; this is a boundary condition
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{condition for time-stepping stability}

\bigskip\medskip
\begin{itemize}
\item SKE: \quad $\displaystyle \frac{\partial s}{\partial t} + u \frac{\partial s}{\partial x} - w = a$

\medskip
\item information about the surface

travels at speed $u$
\item from grid picture, the upwind

scheme can only be stable if
$$|u_j| \Delta t \le \Delta x \quad \text{for every } j \hspace{40mm}$$

    \begin{itemize}
    \item[$\circ$] observed by Courant, Friedrichs, and Lewy in 1928 %\cite{CFL1928}
    \end{itemize}

\vspace{-45mm}
\hfill \includegraphics[width=0.43\textwidth]{cfl.png}

\vspace{15mm}
\item rearranged, this is the \alert{CFL condition} on the time step:
$$\Delta t \le \frac{\Delta x}{\max |u_j|} \hspace{40mm}$$

    \begin{itemize}
    \item[$\circ$] exercise \textbf{12}: add this to \texttt{surface.py} for $u(t,x)$
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{before moving on: SKE versus mass continuity equation}

\begin{itemize}
\item surface kinematical equation (SKE):
\begin{equation*}
\frac{\partial s}{\partial t} - \bu \cdot \bn_s = a
\end{equation*}

    \begin{itemize}
    \item[$\circ$] all quantities are \emph{at surface}, but you must know what $s$ is
    \item[$\circ$] \emph{independent of your model for the glacier below}
    \end{itemize}
\item mass continuity equation:
\begin{equation*}
\frac{\partial H}{\partial t} = M - \Div \bq
\end{equation*}

    \begin{itemize}
    \item[$\circ$] $H=s-b$ is the \emph{ice thickness} (observable if $b$ is)
    \item[$\circ$] $\bq$ is the \emph{ice flux} or \emph{discharge} (not observable)
    \item[$\circ$] $M$ is the \emph{climatic-basal mass balance} (includes internal \& basal)
    \item[$\circ$] \alert{equivalent to SKE $+$ BKE if ice is incompressible} (see notes)
    \end{itemize}
\item \emph{any term} in above equations can be regarded as the \emph{residual}
\item<2> mass continuity equation \emph{with geometrical formula for the flux}:
\begin{equation*}
\frac{\partial H}{\partial t} = M - \Div (\bar \bU H)
\end{equation*}

    \begin{itemize}
    \item[$\circ$] $\bar \bU$ is \emph{vertically-averaged horizontal velocity}
    \end{itemize}
\end{itemize}
\end{frame}


\section[where does the velocity field come from? (hour 2)]{\textbf{where does the velocity field come from?} (hour 2)}

\subsection{basic mathematics}

\begin{frame}{ice in glaciers is a weird fluid}

\begin{itemize}
\item if the ice were
  \begin{itemize}
  \item[$\circ$] faster-moving than it actually is, and
  \item[$\circ$] linearly-viscous
  \end{itemize}

  then it would be a ``typical'' fluid like liquid water
\item for liquid water use the incompressible Navier-Stokes model:
\begin{align*}
\nabla \cdot \mathbf{u} &= 0 &&\text{\emph{incompressibility}} \\
\rho \left(\frac{\partial \bu}{\partial t} + \mathbf{u}\cdot\nabla \mathbf{u}\right) &= -\nabla p + \nabla \cdot \tau_{ij} + \rho \bm{g} &&\text{\emph{stress balance}} \\
2 \nu (D\mathbf{u})_{ij} &= \tau_{ij} &&\text{\emph{flow law}}
\end{align*}

    \begin{itemize}
    \item[$\circ$] $\bu$ is velocity, $p$ is pressure, $D\bu$ is the strain rate tensor, $\tau_{ij}$ is the deviatoric stress tensor, $\rho$ is density, $\bm{g}$ is the acceleration of gravity
    \item[$\circ$] the stress balance equation is ``$m a = F$'', Newton's 2nd law
    \end{itemize}
\item for such fluids: just look it up in computational fluid dynamics (CFD) textbooks, and use the software
\item but that stuff is mostly useless for glaciers \dots
\end{itemize}
\end{frame}


\begin{frame}{glaciology as computational fluid dynamics}

\begin{itemize}
\item numerical glacier modelling is first-class, large-scale CFD
  \begin{itemize}
  \item[$\circ$] \dots\, but it's weird relative to atmosphere and ocean models
  \end{itemize}
\item consider what makes ocean flows exciting:
  \begin{itemize}
  \item[$\circ$] turbulence
  \item[$\circ$] convection
  \item[$\circ$] Coriolis force
  \item[$\circ$] density stratification
  \end{itemize}
\item none of the above list is relevant to ice flow, so CFD textbooks are of limited utility to glaciologists
\item how do we model slow and laminar ice?
\end{itemize}
\end{frame}


\begin{frame}{ice is a slow, shear-thinning fluid}

\begin{itemize}
\item ice fluid is \alert{slow} and \alert{non-Newtonian}
    \begin{itemize}
    \item[$\circ$] ``slow'' is a technical term:
      $$\rho \left(\mathbf{u}_t + \mathbf{u}\cdot\nabla \mathbf{u}\right) \approx 0 \qquad \iff \qquad \begin{pmatrix} \text{forces of inertia} \\ \text{are neglected} \end{pmatrix}$$
    \item[$\circ$] ice is non-Newtonian in a shear-thinning way
        \begin{itemize}
        \item viscosity $\nu$ is not constant, but depends on the fluid state
        \item shear-thinning means higher strain rates implies lower viscosity
        \end{itemize}
    \end{itemize}

\bigskip
\item the standard flow model is \alert{Glen-Steinemann-Nye} (power-law) \alert{Stokes}:
\begin{align*}
\nabla \cdot \mathbf{u} &= 0 &&\text{\emph{incompressibility}} \\
0 &= - \nabla p + \nabla \cdot \tau_{ij} + \rho\, \bm{g} &&\text{\emph{stress balance}} \\
D\mathbf{u}_{ij} &= A \tau^{n-1} \tau_{ij} &&\text{\emph{flow law}}
\end{align*}

    \begin{itemize}
    \item[$\circ$] $\rho$ is ice density, $\bm{g}$ is acceleration of gravity, $A$ is ice softness
    \end{itemize}

\bigskip
\item the standard glacier geometry model is \alert{SKE $+$ Stokes}
\end{itemize}
\end{frame}


\begin{frame}{regarding spatial derivatives}

\begin{itemize}
\item the Stokes equations are loaded with spatial derivatives:
{\small
\begin{align*}
\nabla \cdot \bu &= 0 \\
0 &= - \nabla p + \nabla \cdot \tau_{ij} + \rho\, \bm{g} \\
(D\bu)_{ij} &= A \tau^{n-1} \tau_{ij}
\end{align*}
}
\item $p$ is a scalar, $\bu$ is vector, and $(D\bu)_{ij},\tau_{ij}$ are tensors (matrices)
\item by definition, in 3D coordinates $(x_1,x_2,x_3)$ with $\bu=\left<u_1,u_2,u_3\right>$:
\begin{align*}
\nabla p &= \left<\frac{\partial p}{\partial x_1}, \frac{\partial p}{\partial x_2}, \frac{\partial p}{\partial x_3}\right> &&\text{\emph{vector}} \\
\nabla \cdot \bu &= \frac{\partial u_1}{\partial x_1} + \frac{\partial u_2}{\partial x_2} + \frac{\partial u_3}{\partial x_3} &&\text{\emph{scalar}} \\
\nabla \cdot \tau_{ij} &= \left<\nabla \cdot \tau_{1}, \nabla \cdot \tau_{2}, \nabla \cdot \tau_{3}\right> &&\text{\emph{vector}} \\
(D\bu)_{ij} &= \frac{1}{2}\left(\grad \bu + (\grad \bu)^\top\right)_{ij} &&\text{\emph{tensor}} \\
\tau^{n-1} &= \big(\frac{1}{2}\tau_{ij}\,\tau_{ji}\big)^{(n-1)/2} &&\text{\emph{scalar}}
\end{align*}
\item getting comfortable with such notation is a skill in itself
\end{itemize}
\end{frame}


\begin{frame}{``slow'' means no memory of velocity/momentum}

\begin{itemize}
\item note something simple: \alert{no time derivatives} in the Stokes model!
\begin{align*}
\nabla \cdot \bu &= 0 \\
0 &= - \nabla p + \nabla \cdot \tau_{ij} + \rho\, \bm{g} \\
(D\bu)_{ij} &= A \tau^{n-1} \tau_{ij}
\end{align*}
\item the velocity and pressure fields of a glacier are \alert{instantaneous functions} of the glacier geometry and its boundary stresses
\item thus a time-stepping ice sheet simulation will \alert{recompute} the pressure and velocity fields \alert{at every time step}
  \begin{itemize}
  \item[$\circ$] velocity from the previous time step is \emph{not} required, unlike in a weather model for example
  \item[$\circ$] for glaciers, velocity is a ``diagnostic'' output, not needed for starting or restarting a model run
  \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{plane flow Stokes}

\begin{itemize}
\item let's return to the $x,z$ flow-line plane, and look at details
\item the $n=3$ Glen-law Stokes equations say:
\begin{empheq}[]{align}
u_x + w_z &= 0 &&\text{\emph{incompressibility}}\notag \\
0 &= -p_x + \tau_{11,x} + \tau_{13,z} &&\text{\emph{stress balance} ($x$)} \notag \\
0 &= -p_z + \tau_{13,x} - \tau_{11,z} - \rho g &&\text{\emph{stress balance} ($z$)} \notag \\
u_x &= A \tau^2 \tau_{11} &&\text{\emph{flow law (diagonal)}}\notag \\
\frac{1}{2} \left(u_z + w _x\right) &= A \tau^2 \tau_{13} &&\text{\emph{flow law (off-diagonal)}} \notag
\end{empheq}

    \begin{itemize}
    \item[$\circ$] here the subscripts $x,z$ denote partial derivatives
    \item[$\circ$] $\tau_{11}$ and $\tau_{33}$ are (deviatoric) longitudinal stresses
        \begin{itemize}
        \item observation: $\tau_{33}=-\tau_{11}$ \comm{why?}
        \end{itemize}
    \item[$\circ$] $\tau_{13}$ is the ``vertical'' shear stress
    \item[$\circ$] after simplification: \, $\tau^2 = \tau_{11}^2 + \tau_{13}^2$
    \end{itemize}
\item we have 5 equations in 5 unknowns ($u,w,p,\tau_{11},\tau_{13}$)
\item this is complicated enough \dots \,try a simplified situation?
\end{itemize}
\end{frame}

\subsection{some numerics}

\begin{frame}{SIA for velocity}

\begin{itemize}
\item the planar, isothermal, Glen-law, non-sliding \alert{shallow ice approximation (SIA)} is the following system of equations:
\begin{gather*}
u = - \frac{2}{n+1} A (\rho g)^n \left[\big(s - b\big)^{n+1} - \big(s-z\big)^{n+1}\right] \left|\frac{\partial s}{\partial x}\right|^{n-1} \frac{\partial s}{\partial x} \\
\frac{\partial u}{\partial x} + \frac{\partial w}{\partial z} = 0
\end{gather*}

  \begin{itemize}
  \item[$\circ$] $u,w$ depend on $t,x,z$, but $s$ depends only on $t,x$
  \end{itemize}

\item note that the horizontal velocity formula is of the form
	$$u = - C\, \frac{\partial s}{\partial x},$$
with positive coefficient $C>0$, so \alert{ice flows downhill}
\end{itemize}
\end{frame}


\begin{frame}{slopes and elevations onto the staggered grid}

\begin{center}
\includegraphics[width=0.4\textwidth]{staggered}
\end{center}

\begin{itemize}
\item center-differencing the surface elevations gives \alert{slopes computed on the staggered grid}:
    $$\frac{\partial s}{\partial x}\Big|_{j+1/2} = \frac{s_{j+1} - s_j}{\Delta x}$$
\item averaging gives \alert{elevations computed on the staggered grid}:
    $$s_{j+1/2} = \frac{s_j + s_{j+1}}{2}, \qquad b_{j+1/2} = \frac{b_j + b_{j+1}}{2}$$
\end{itemize}
\end{frame}


\begin{frame}[fragile]
\frametitle{Python code: finite differences for surface horizontal velocity}

\begin{itemize}
\item recall the SKE needs only the \emph{surface value} of velocity:
    $$u_{j+1/2} = - C\, \left(\frac{H_j + H_{j+1}}{2}\right)^{n+1} \left|\frac{s_{j+1}-s_j}{\Delta x}\right|^{n-1} \frac{s_{j+1}-s_j}{\Delta x}$$
where $C = \frac{2}{n+1} A (\rho g)^n$ is constant, and $H=s-b$
\item computing this at staggered-grid locations is a Python function:
\begin{lstlisting}[language=PythonPlus]
def u(x, b, s):
    dx = x[1] - x[0]
    dsdx = (s[1:] - s[:-1]) / dx
    Hstag = ((s[:-1] - b[:-1]) + (s[1:] - b[1:])) / 2.0
    C_u = (2.0 / (n + 1)) * A * (rhoi * g) ** n
    return -C_u * Hstag ** (n + 1) * abs(dsdx) ** (n - 1) * dsdx
\end{lstlisting}
\end{itemize}
\end{frame}


\begin{frame}{surface vertical velocity (and ice flux)}

\begin{itemize}
\item we can recover the vertical velocity $w$ from \alert{incompressibility}

  \begin{itemize}
  \item[$\circ$] recall: \,$\frac{\partial u}{\partial x} + \frac{\partial w}{\partial z} = 0$
  \item[$\circ$] integrate vertically, using non-penetrating ($w(t,x,b(x))=0$), then incompressibility, then Leibniz rule, then non-sliding ($u(t,x,b(x))=0$), to get $w$ at the surface:
\begin{align*}
\alert{w(t,x,s(t,x))} &= - \int_{b(x)}^{s(t,x)} \frac{\partial u}{\partial x}(t,x,z)\,dz \\
     &= \alert{u(x,s(t,x)) \frac{\partial s}{\partial x}(t,x) - \frac{\partial q}{\partial x}(t,x)}
\end{align*}
  \item[$\circ$] exercises \textbf{4} and \textbf{5} help with the Leibniz rule
  \end{itemize}

\medskip
\item here \, $\ds \alert{q(t,x) = \int_{b(x)}^{s(t,x)} u(t,x,z)\,dz}$\, is the \alert{ice flux}
  \begin{itemize}
  \item[$\circ$] alternate form:  $q = \bar u H$ where $\bar u$ is vertically-averaged horizontal velocity and $H$ is ice thickness
  \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{ice flux and diffusion}

\begin{itemize}
\item in the planar case the ice flux is \,$\ds q = \int_b^s u\,dz$
\item vertically-integrating the horizontal velocity gives
    $$q  = - \Dsia \frac{\partial s}{\partial x}$$
where the (effective) \alert{diffusivity} is
    $$\Dsia = \frac{2}{n+2} A (\rho g)^n H^{n+2} \left|\frac{\partial s}{\partial x}\right|^{n-1}$$
\item the formula for surface vertical velocity also uses it:
    $$w = u \frac{\partial s}{\partial x} + \frac{\partial}{\partial x} \left(\Dsia \frac{\partial s}{\partial x}\right)$$

  \begin{itemize}
  \item[$\circ$] we will use this form to compute $w$ at the surface
  \end{itemize}
\item we will return to what diffusivity \emph{means}, but for now we apply finite differences to compute $\Dsia$ and $w$ in an example
\end{itemize}
\end{frame}


\begin{frame}[fragile]
\frametitle{finite differences for $\Dsia$ and $w$}

\begin{itemize}
\item for $\Dsia$ on the staggered grid:
    $$\Dsia_{j+1/2} = \frac{2}{n+2} A (\rho g)^n \left(\frac{H_j + H_{j+1}}{2}\right)^{n+2} \left|\frac{s_{j+1}-s_j}{\Delta x}\right|^{n-1}$$
\begin{lstlisting}[language=PythonPlus]
def diffusiveD(x, b, s):
    dx = x[1] - x[0]
    dsdx = (s[1:] - s[:-1]) / dx
    Hstag = ((s[:-1] - b[:-1]) + (s[1:] - b[1:])) / 2.0
    C_D = (2.0 / (n + 2)) * A * (rhoi * g) ** n
    return C_D * Hstag ** (n + 2) * abs(dsdx) ** (n - 1)
\end{lstlisting}
\item for $w$, let $\phi_{j+1/2} = u_{j+1/2} \frac{s_{j+1}-s_j}{\Delta x}$ on staggered grid, so:
    $$w_j = \frac{\phi_{j-1/2} + \phi_{j+1/2}}{2} + \frac{\Dsia_{j+1/2} (s_{j+1}-s_j) - \Dsia_{j-1/2} (s_{j}-s_{j-1})}{\Delta x^2}$$

  \begin{itemize}
  \item[$\circ$] we get $w_j$ on the regular grid (Python function not shown)
  \end{itemize}

\bigskip
\item formulas for $u,\Dsia,w$ are all in \href{https://github.com/bueler/mccarthy/blob/master/py/shallow.py}{py/shallow.py}
\end{itemize}
\end{frame}

\begin{frame}{time-dependent SKE $+$ SIA model}

\begin{itemize}
\item here are equations, so far, of the planar, non-sliding, and isothermal \alert{time-dependent model} combining SKE and SIA:
{\footnotesize
\begin{align*}
\frac{\partial s}{\partial t} &= a + w - u \frac{\partial s}{\partial x} \\
u &= - \frac{2}{n+1} A (\rho g)^n \big(s - b\big)^{n+1} \left|\frac{\partial s}{\partial x}\right|^{n-1} \frac{\partial s}{\partial x} \hspace{2mm} (\text{\emph{surface value}}) \\
w &= u \frac{\partial s}{\partial x} - \frac{\partial q}{\partial x}  \hspace{35mm} (\text{\emph{surface value}})\\
q &= - \Dsia \frac{\partial s}{\partial x}
\end{align*}
}

\item we can simplify to just this for $s(t,x)$:
    $$\alert{\frac{\partial s}{\partial t} = \frac{\partial}{\partial x}\left(\Dsia \frac{\partial s}{\partial x}\right) + a}$$

  \begin{itemize}
  \item[$\circ$] along with the formula: \, $\Dsia = \frac{2}{n+2} A (\rho g)^n H^{n+2} \left|\frac{\partial s}{\partial x}\right|^{n-1}$
  \item[$\circ$] $u,w,q$ are not directly needed to express geometry evolution
  \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{time-dependent SKE $+$ SIA model as diffusion equation}

\begin{itemize}
\item the SKE $+$ SIA geometry evolution model is
    $$\frac{\partial s}{\partial t} = \frac{\partial}{\partial x}\left(\Dsia \frac{\partial s}{\partial x}\right) + a$$
\item this model \alert{diffuses} the surface elevation
  \begin{itemize}
  \item[$\circ$] because ice flows downhill: $\ds q = - \Dsia \frac{\partial s}{\partial x}$
  \end{itemize}
\item analogy to Fourier's heat equation:
$$\frac{\partial s}{\partial t} = \frac{\partial}{\partial x}\left(\Dsia \frac{\partial s}{\partial x}\right) + a \quad \leftrightarrow \quad \frac{\partial T}{\partial t} = \frac{\partial}{\partial x}\left(D \frac{\partial T}{\partial x}\right) + f$$

  \begin{itemize}
  \item[$\circ$] $T$ is temperature and $f$ is a heat source
  \item[$\circ$] Fourier's law says heat flux is ``downtemperature'': $\ds q = - D \frac{\partial T}{\partial x}$
  \end{itemize}
\item but the SKE $+$ SIA is extremely nonlinear because
    $$\Dsia = \frac{2}{n+2} A (\rho g)^n H^{n+2} \left|\frac{\partial s}{\partial x}\right|^{n-1}$$
\end{itemize}
\end{frame}

\begin{frame}{time-dependent SKE $+$ Stokes}

\begin{itemize}
\item what is the \alert{standard model} of glaciology?
\item I think it is \alert{SKE $+$ Stokes with admissible surface}:
\begin{align*}
\frac{\partial s}{\partial t} - \mathbf{u} \cdot \mathbf{n}_s &= a && \mcomm{SKE} \\
\nabla \cdot \mathbf{u} &= 0 && \mcomm{incompressibility} \\
0 &= - \nabla p + \nabla \cdot \tau_{ij} + \rho\, \bm{g} && \mcomm{stress balance} \\
D\mathbf{u}_{ij} &= A \tau^{n-1} \tau_{ij} && \mcomm{flow law} \\
s &\ge b && \mcomm{admissibility}
\end{align*}
\item implementing well is a topic of current research
  \begin{itemize}
  \item[$\circ$] reliable and fast coupled solvers are not yet available
  \end{itemize}
\end{itemize}

    \begin{itemize}
    \item[$\circ$] there are no mass sources within $\Omega$
    \item[$\circ$] there is an interval of time $t$ so that \,$z_0-\eps < s(t,x) < z_0 + \eps$
    %\item[$\circ$] ice has constant density $\rhoi$
    \item[$\circ$] the ice velocity $\bu(t,x,z)=\left<u,w\right>$ is defined on $\{z<s(t,x)\}$
    \item[$\circ$] the SMB $a(t,x)$ is a mass flux only across $\Gamma_t$:
        \begin{itemize}
        \item $\bpsi\big|_{\Gamma_t} = - \rho_i a(t,x) \hbz$  \comm{note $a>0$ is precipitation}
        \end{itemize}
    \item[$\circ$] the other mass fluxes are kinematic (advective):
        \begin{itemize}
        \item $\bpsi\big|_{\Gamma_b} = \rho_i w(t,x,z_0-\eps) \hbz$
        \item $\bpsi\big|_{\Gamma_{\{\ell,r\}}} = \begin{cases} \rho_i u(t,x_{\{\ell,r\}},z) \hbx, & \text{if } z < s(t,x_{\{\ell,r\}}), \\ \bzero, & \text{otherwise} \end{cases}$
        \end{itemize}
    \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{SKE from mass conservation 3}

\begin{itemize}
\item mass is conserved:
\begin{align*}
\frac{dM}{dt} &= - \int_{\partial \Omega} \bpsi \cdot \hbn\,ds \\
  &= \underbrace{\rhoi \int_{x_0-\delta}^{x_0+\delta} a(t,x)\,dx}_{\Gamma_t} + \underbrace{\rhoi \int_{x_0-\delta}^{x_0+\delta} w(t,x,z_0-\eps)\,dx}_{\Gamma_b} \\
  &\quad + \underbrace{\rhoi \int_{z_0-\eps}^{s(t,x_0-\delta)} u(t,x_0-\delta,z)\,dz}_{\Gamma_\ell} - \underbrace{\rhoi \int_{z_0-\eps}^{s(t,x_0+\delta)} u(t,x_0+\delta,z)\,dz}_{\Gamma_r}
\end{align*}

\end{itemize}
\end{frame}


\begin{frame}{SKE from mass conservation 4}

\begin{itemize}
\item time derivative:
\begin{align*}
\frac{dM}{dt} &= \lim_{\omega\to 0} \frac{M(t+\omega) - M(t)}{\omega} \\
    &= \lim_{\omega\to 0} \frac{\rhoi}{\omega} \int_\Omega \mathbb{1}_{\{z<s(t+\omega,x)\}} - \mathbb{1}_{\{z<s(t,x)\}}\,dz\,dx \\
    &= \lim_{\omega\to 0} \frac{\rhoi}{\omega} \int_{x_0-\delta}^{x_0+\delta} s(t+\omega,x) - s(t,x)\,dx \\
    &= \rho_i \int_{x_0-\delta}^{x_0+\delta} \frac{\partial s}{\partial t}(t,x)\,dx
\end{align*}
\item assume smoothness sufficient for Taylor expansions on $\Omega$:
\begin{align*}
a(t,x) &= a(t,x_0) + O(\delta) \\
u(t,x,z) &= u(t,x_0,z_0) + O(\delta) + O(\eps) \\
w(t,x,z) &= w(t,x_0,z_0) + O(\delta) + O(\eps) \\
\frac{\partial s}{\partial t}(t,x) &= \frac{\partial s}{\partial t}(t,x_0) + O(\delta)
\end{align*}
\end{itemize}
\end{frame}


\begin{frame}{SKE from mass conservation 5}

\begin{itemize}
\item combine mass conservation, time derivative, and Taylor:
\begin{align*}
2\delta \rhoi \frac{\partial s}{\partial t}(t,x_0) + O(\delta^2) &= 2\delta \rhoi a(t,x_0) + O(\delta^2) \\
  &\quad + 2\delta \rhoi w(t,x_0,z_0) + O(\delta^2) + O(\delta \eps) \\
  &\quad - \rho_i u(t,x_0,z_0) \Big[s(t,x_0+\delta) - s(t,x_0-\delta)\Big] + O(\delta \eps)
\end{align*}
\item divide by $2\rhoi\delta$:
\begin{align*}
\frac{\partial s}{\partial t}(t,x_0) &= a(t,x_0) + w(t,x_0,z_0) - u(t,x_0,z_0) \frac{s(t,x_0+\delta) - s(t,x_0-\delta)}{2\delta} \\
  &\quad  + O(\delta) + O(\eps)
\end{align*}
\item limit $\delta, \eps \to 0$ to get SKE at $(t,x_0,z_0) = (t,x_0,s(t,x_0))$:
  $$\frac{\partial s}{\partial t} = a + w - u \frac{\partial s}{\partial x}$$
\end{itemize}
\end{frame}
